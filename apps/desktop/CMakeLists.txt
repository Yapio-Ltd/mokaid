cmake_minimum_required(VERSION 3.30)
set(MOKAID_VERSION "0.1.0" CACHE STRING "Numeric application version")
if(NOT MOKAID_VERSION MATCHES "^[0-9]+\\.[0-9]+\\.[0-9]+$")
  message(FATAL_ERROR "MOKAID_VERSION must be X.Y.Z")
endif()
project(MokaidDesktop VERSION ${MOKAID_VERSION} LANGUAGES CXX)
set(MOKAID_RELEASE_VERSION "${MOKAID_VERSION}" CACHE STRING "Release version including prerelease suffix")
set(MOKAID_RELEASE_CHANNEL "stable" CACHE STRING "Release channel: stable or beta")
if(NOT MOKAID_RELEASE_CHANNEL MATCHES "^(stable|beta)$")
  message(FATAL_ERROR "Unsupported release channel")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
option(MOKAID_BUILD_APP "Build the native desktop application" ON)
option(MOKAID_ENABLE_UPDATES "Include platform update frameworks" OFF)
option(MOKAID_ENABLE_SANITIZERS "Enable address and undefined behavior sanitizers" OFF)
option(MOKAID_BETA "Use a separate beta application identity" OFF)
option(MOKAID_DEVELOPMENT "Use an isolated development identity, without update feeds" OFF)
if(MOKAID_DEVELOPMENT AND (MOKAID_BETA OR MOKAID_ENABLE_UPDATES))
  message(FATAL_ERROR "Development builds cannot use beta identity or signed update feeds")
endif()
if(MOKAID_BETA)
  set(MOKAID_RELEASE_CHANNEL "beta" CACHE STRING "Release channel" FORCE)
endif()
set(MOKAID_QT_VERSION "6.11.2" CACHE STRING "Exact supported Qt version")
set(MOKAID_API_ORIGIN "https://mokaid.com" CACHE STRING "Trusted API origin")
set(MOKAID_WEB_ORIGIN "${MOKAID_API_ORIGIN}" CACHE STRING "Trusted browser sign-in origin (explicitly separate for local development)")
if(APPLE)
  enable_language(OBJCXX)
  set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS" FORCE)
endif()
include(CTest)
add_library(mokaid_options INTERFACE)
if(MSVC)
  target_compile_options(mokaid_options INTERFACE /W4 /permissive- /utf-8)
else()
  target_compile_options(mokaid_options INTERFACE -Wall -Wextra -Wpedantic)
  if(MOKAID_ENABLE_SANITIZERS)
    target_compile_options(mokaid_options INTERFACE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(mokaid_options INTERFACE -fsanitize=address,undefined)
  endif()
endif()
add_subdirectory(core)
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/engine/CMakeLists.txt")
  add_subdirectory(engine)
endif()
if(MOKAID_BUILD_APP)
  if(NOT APPLE AND NOT WIN32)
    message(FATAL_ERROR "Desktop supports macOS and Windows. Use MOKAID_BUILD_APP=OFF for portable tests.")
  endif()
  find_package(Qt6 ${MOKAID_QT_VERSION} EXACT REQUIRED COMPONENTS Core Gui Quick QuickControls2 Network WebSockets Sql Concurrent WebEngineQuick Test QuickTest)
  qt_standard_project_setup(REQUIRES 6.8)
  add_subdirectory(platform)
  add_subdirectory(storage)
  add_subdirectory(network)
  add_subdirectory(application)
  add_subdirectory(preview)
  add_subdirectory(renderer)
  add_subdirectory(bridge)
  add_subdirectory(presentation)
  add_subdirectory(app)
  if(BUILD_TESTING)
    add_subdirectory(tests/graphics)
  endif()
endif()
