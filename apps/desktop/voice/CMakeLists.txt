find_package(Qt6 ${MOKAID_QT_VERSION} EXACT REQUIRED COMPONENTS Multimedia TextToSpeech Concurrent)
find_package(Python3 3.11 REQUIRED COMPONENTS Interpreter)
option(MOKAID_BUNDLE_LOCAL_VOICE "Bundle verified local voice engines and model weights" ON)
set(MOKAID_VOICE_RUNTIME_DIR "${CMAKE_BINARY_DIR}/voice-runtime" CACHE PATH "Provisioned voice runtime directory")
add_library(mokaid_voice STATIC voice_controller.cpp audio_utils.cpp
    include/mokaid/voice/voice_controller.hpp include/mokaid/voice/audio_utils.hpp)
target_include_directories(mokaid_voice PUBLIC include)
target_link_libraries(mokaid_voice PUBLIC Qt6::Core Qt6::Multimedia PRIVATE Qt6::TextToSpeech Qt6::Concurrent mokaid_options)
target_compile_definitions(mokaid_voice PRIVATE MOKAID_VOICE_DEV_RUNTIME="${MOKAID_VOICE_RUNTIME_DIR}")
if(MOKAID_BUNDLE_LOCAL_VOICE)
    if(APPLE)
        set(voice_arch "${CMAKE_OSX_ARCHITECTURES}")
        if(NOT voice_arch)
            set(voice_arch "${CMAKE_SYSTEM_PROCESSOR}")
        endif()
        if(voice_arch STREQUAL "x86_64")
            set(voice_platform macos-x64)
        elseif(voice_arch STREQUAL "arm64")
            set(voice_platform macos-arm64)
        else()
            message(FATAL_ERROR "Local voice requires a single macOS arm64 or x86_64 target architecture")
        endif()
    elseif(WIN32 AND CMAKE_SIZEOF_VOID_P EQUAL 8)
        set(voice_platform windows-x64)
    else()
        message(FATAL_ERROR "Bundled local voice currently supports macOS arm64/x64 and Windows x64")
    endif()
    set(voice_ninja_args)
    if(CMAKE_GENERATOR STREQUAL "Ninja")
        list(APPEND voice_ninja_args --ninja "${CMAKE_MAKE_PROGRAM}")
    endif()
    add_custom_command(OUTPUT "${MOKAID_VOICE_RUNTIME_DIR}/manifest.json"
        COMMAND Python3::Interpreter "${CMAKE_CURRENT_SOURCE_DIR}/provision.py"
            --output "${MOKAID_VOICE_RUNTIME_DIR}" --cache "${CMAKE_BINARY_DIR}/voice-downloads"
            --platform "${voice_platform}" --cmake "${CMAKE_COMMAND}" ${voice_ninja_args}
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/provision.py"
        COMMENT "Provisioning verified local Whisper and Kokoro voice models" VERBATIM)
    add_custom_target(mokaid_voice_runtime DEPENDS "${MOKAID_VOICE_RUNTIME_DIR}/manifest.json")
endif()
function(mokaid_bundle_voice target)
    if(NOT MOKAID_BUNDLE_LOCAL_VOICE)
        return()
    endif()
    add_dependencies(${target} mokaid_voice_runtime)
    set_property(TARGET ${target} APPEND PROPERTY LINK_DEPENDS "${MOKAID_VOICE_RUNTIME_DIR}/manifest.json")
    if(APPLE)
        set(destination "$<TARGET_BUNDLE_DIR:${target}>/Contents/Resources/voice")
        install(DIRECTORY "${MOKAID_VOICE_RUNTIME_DIR}/" DESTINATION "Mokaid.app/Contents/Resources/voice" USE_SOURCE_PERMISSIONS)
    else()
        set(destination "$<TARGET_FILE_DIR:${target}>/voice")
        install(DIRECTORY "${MOKAID_VOICE_RUNTIME_DIR}/" DESTINATION voice USE_SOURCE_PERMISSIONS)
    endif()
    add_custom_command(TARGET ${target} POST_BUILD COMMAND "${CMAKE_COMMAND}" -E copy_directory
        "${MOKAID_VOICE_RUNTIME_DIR}" "${destination}" VERBATIM)
endfunction()
