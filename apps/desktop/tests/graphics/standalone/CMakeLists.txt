cmake_minimum_required(VERSION 3.30)
project(MokaidGraphicsProbe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
set(BUILD_TESTING OFF CACHE BOOL "The standalone build runs the probe explicitly")
set(MOKAID_ENABLE_UPDATES OFF CACHE BOOL "The probe never checks for updates")
option(MOKAID_ENABLE_SANITIZERS "Enable ASan and UBSan in the integration probe" OFF)
set(MOKAID_SANITIZER_SET "address,undefined" CACHE STRING "Supported probe sanitizer combination")
set_property(CACHE MOKAID_SANITIZER_SET PROPERTY STRINGS "address,undefined" "undefined")
if(NOT MOKAID_SANITIZER_SET MATCHES "^(address,undefined|undefined)$")
    message(FATAL_ERROR "Unsupported integration-probe sanitizer combination")
endif()
if(APPLE)
    enable_language(OBJCXX)
    set(CMAKE_OSX_DEPLOYMENT_TARGET "13.0" CACHE STRING "Minimum macOS" FORCE)
endif()
find_package(Qt6 6.11.2 EXACT REQUIRED COMPONENTS Core Gui Quick QuickControls2
    Network WebSockets Sql Concurrent WebEngineQuick Test)
qt_standard_project_setup(REQUIRES 6.8)
add_library(mokaid_options INTERFACE)
if(MSVC)
    target_compile_options(mokaid_options INTERFACE /W4 /permissive- /utf-8)
else()
    target_compile_options(mokaid_options INTERFACE -Wall -Wextra -Wpedantic)
    if(MOKAID_ENABLE_SANITIZERS)
        # Instrument every source module in this standalone build, including the
        # Qt-independent engine and Objective-C++ renderer, not only the harness.
        add_compile_options("-fsanitize=${MOKAID_SANITIZER_SET}" -fno-omit-frame-pointer)
        add_link_options("-fsanitize=${MOKAID_SANITIZER_SET}")
    endif()
endif()
get_filename_component(desktop "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
foreach(module IN ITEMS core engine platform storage network application preview renderer bridge)
    add_subdirectory("${desktop}/${module}" "modules/${module}")
endforeach()
add_subdirectory(.. graphics)
